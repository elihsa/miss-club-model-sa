# Then in your CMS loader:
document.querySelectorAll('[data-cms-key]').forEach(el => {
  const key = el.getAttribute('data-cms-key');
  if (content[key] && key.startsWith('color')) {
    document.documentElement.style.setProperty('--' + key.replace(/([A-Z])/g, '-$1').toLowerCase(), content[key]);
  }
});
